  }).join("");

  show("sc-load");

  var idx=0,total=STAGES.length;
  function step(){
    var items=stagesEl.querySelectorAll(".stage");
    if(idx>0){items[idx-1].classList.remove("is-now");items[idx-1].classList.add("is-done");
      items[idx-1].querySelector(".stage__dot").innerHTML='<svg width="10" height="10" viewBox="0 0 10 10" fill="none"><path d="M1.5 5.2l2.3 2.3L8.5 2.6" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"/></svg>';}
    if(idx>=total){
      fill.style.width="100%";
      setTimeout(function(){show("sc-report");renderReport();},520);
      return;
    }
    items[idx].classList.add("is-now");
    fill.style.width=Math.round(((idx+0.5)/total)*100)+"%";
    idx++;
    setTimeout(step, idx===1?700:(560+Math.random()*520));
  }
  setTimeout(step,380);
}

function gaugeSVG(score){
  var r=92,cx=110,cy=118,len=Math.PI*r;
  var ticks="";
  [0,.25,.5,.75,1].forEach(function(f){
    var a=Math.PI*(1-f);
    var x1=cx+104*Math.cos(a),y1=cy-104*Math.sin(a);
    var x2=cx+113*Math.cos(a),y2=cy-113*Math.sin(a);
    ticks+='<path d="M'+x1.toFixed(1)+' '+y1.toFixed(1)+'L'+x2.toFixed(1)+' '+y2.toFixed(1)+'" stroke="#C3B394" stroke-width="2" stroke-linecap="round"/>';
  });
  return '<svg viewBox="0 0 220 132" role="img" aria-label="'+score+'/100">'+
    ticks+
    '<path d="M18 118 A92 92 0 0 1 202 118" fill="none" stroke="#E4D9C2" stroke-width="15" stroke-linecap="round"/>'+
    '<path class="gauge-arc" d="M18 118 A92 92 0 0 1 202 118" fill="none" stroke="'+bandColor(score)+'" stroke-width="15" stroke-linecap="round" stroke-dasharray="'+len.toFixed(1)+'" stroke-dashoffset="'+len.toFixed(1)+'" data-len="'+len.toFixed(1)+'" data-score="'+score+'"/>'+
  '</svg>';
}

function deltaHTML(cur,prev){
  var d=cur-prev;
  if(d>0)return '<span class="delta delta--up">▲ '+fmt(d)+'</span>';
  if(d<0)return '<span class="delta delta--down">▼ '+fmt(Math.abs(d))+'</span>';
  return '<span class="delta delta--flat">— 0</span>';
}

function themeRow(item,max,type,locked){
  var pct=Math.max(6,Math.round(item.n/max*100));
  return '<li class="theme theme--'+type+(locked?' is-locked':'')+'"'+(locked?' aria-hidden="true"':'')+'>'+
    '<div class="theme__top"><span class="theme__t">'+esc(item.t)+'</span>'+
    '<span class="theme__c"><b>'+fmt(item.n)+'</b> '+t("rpt.inreview")+'</span></div>'+
    '<div class="theme__meter"><i style="width:'+pct+'%"></i></div>'+
    '<blockquote class="theme__q">“'+esc(item.q)+'”<footer>'+esc(item.by)+' · '+t("rpt.orig")+'</footer></blockquote>'+
    '<span class="theme__lock"><svg width="11" height="11" viewBox="0 0 12 12" fill="none" aria-hidden="true"><rect x="2" y="5.4" width="8" height="5.6" rx="1" stroke="currentColor" stroke-width="1.3"/><path d="M3.8 5.4V3.8a2.2 2.2 0 014.4 0v1.6" stroke="currentColor" stroke-width="1.3"/></svg>'+t("rpt.lockpill")+'</span>'+
  '</li>';
}

function animateReport(){
  var arc=document.querySelector(".gauge-arc");
  if(arc){
    var len=parseFloat(arc.getAttribute("data-len")),sc=parseFloat(arc.getAttribute("data-score"));
    requestAnimationFrame(function(){
      setTimeout(function(){
        arc.style.transition="stroke-dashoffset 1.3s cubic-bezier(.2,.7,.3,1)";
        arc.style.strokeDashoffset=(len-(len*sc/100)).toFixed(1);
      },140);
    });
  }
  var num=document.getElementById("scoreNum");
  if(num){
    var target=state.data.score,start=null,dur=1200;
    function tick(ts){
      if(!start)start=ts;
      var p=Math.min((ts-start)/dur,1);
      var e=1-Math.pow(1-p,3);
      num.innerHTML=Math.round(target*e)+'<small>'+t("rpt.score.of")+'</small>';
      if(p<1)requestAnimationFrame(tick);
    }
    requestAnimationFrame(tick);
  }
  var bars=document.querySelectorAll("#splitBar i");
  bars.forEach(function(el,i){setTimeout(function(){el.style.width=el.getAttribute("data-w")+"%";},260+i*130);});

  var io=new IntersectionObserver(function(entries){
    entries.forEach(function(en){
      if(en.isIntersecting){en.target.classList.add("in");io.unobserve(en.target);}
    });
  },{threshold:.12,rootMargin:"0px 0px -40px 0px"});
  document.querySelectorAll(".reveal").forEach(function(el,i){
    el.style.transitionDelay=(i%6)*70+"ms";
    io.observe(el);
  });
}

      }
      err.hidden=true;state.email=v;state.unlocked=true;
      var lw=document.getElementById("lockwall");
      lw.classList.add("is-gone");
      setTimeout(function(){renderReport();window.scrollTo({top:Math.max(0,lw.getBoundingClientRect().top+window.scrollY-140),behavior:"smooth"});},380);
    });
  }
  var tb=document.getElementById("tickBtn");

    tb.addEventListener("click",function(){

      tb.setAttribute("aria-pressed",state.done?"true":"false");
      document.getElementById("tickLabel").textContent=state.done?t("rpt.act.donet"):t("rpt.act.done");
    });
  }

      var ta=document.createElement("textarea");ta.value=url;document.body.appendChild(ta);ta.select();
      try{document.execCommand("copy");ok();}catch(e){toast(t("rpt.copyfail"));}
      document.body.removeChild(ta);
    }
  });
  var nb=document.getElementById("newBtn");
  if(nb)nb.addEventListener("click",function(){reset();});
}
